# Realtime P0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the production WebSocket genuinely connect through Nginx and replace the iOS three-second fallback loop with confirmed-connection state and 30/60/120/300-second fallback intervals.

**Architecture:** Refactor the existing `RealtimeHub` just enough to inject authentication and heartbeat timing in tests. Extract pure Swift realtime policy/state into files shared by the app and a Swift Package test target. Check in the exact production Nginx server block and a token-from-environment smoke script, then deploy with backup, config test, reload, and rollback commands.

**Tech Stack:** Node.js built-in test runner, `ws`, Swift 5 mode on iOS 17, Swift Package Manager tests, Nginx, systemd.

## Global Constraints

- Do not change WebSocket protocol payloads; `sync.available` remains backward compatible.
- Mark iOS realtime connected only after a successful WebSocket ping callback or a received valid event.
- Fallback delays are exactly 30, 60, 120, and 300 seconds, capped at 300 seconds; a successful fallback sync resets to 30 seconds.
- Foreground activation still triggers one immediate incremental sync.
- No production file changes before saving `/etc/nginx/sites-enabled/zbt-backend` and passing `nginx -t`.
- No token value appears in command arguments, source, logs, or Git.

Implement this public shape while preserving the current connection, heartbeat, and payload logic:

```js
export class RealtimeHub {
  constructor({ authenticate = userForToken, heartbeatIntervalMs = 30000, logger: eventLogger = logger } = {}) {
    this.authenticate = authenticate;
    this.heartbeatIntervalMs = heartbeatIntervalMs;
    this.logger = eventLogger;
    this.connectionsByUserID = new Map();
    this.wss = new WebSocketServer({ noServer: true });
    this.heartbeat = null;
    this.wss.on("connection", (socket, _request, user) => this.addConnection(user.id, socket));
  }

  close() {
    if (this.heartbeat) clearInterval(this.heartbeat);
    this.heartbeat = null;
    for (const socket of this.wss.clients) socket.terminate();
    this.wss.close();
    this.connectionsByUserID.clear();
  }
}

export function createRealtimeHub(options = {}) {
  return new RealtimeHub(options);
}

export const realtimeHub = createRealtimeHub();

export function attachRealtime(server, options = undefined) {
  const hub = options ? createRealtimeHub(options) : realtimeHub;
  hub.attach(server);
  return hub;
}
```

Inside `attach`, replace direct `userForToken(token)` with `this.authenticate(token)`, use `this.heartbeatIntervalMs`, and use `this.logger` for connection events.

- [ ] **Step 3: Implement the pure policy and state**

```swift
import Foundation

public struct RealtimeFallbackPolicy: Sendable {
    private let delays: [UInt64] = [30, 60, 120, 300]

    public init() {}

    public func delayNanoseconds(forAttempt attempt: Int) -> UInt64 {
        let index = min(max(0, attempt), delays.count - 1)
        return delays[index] * 1_000_000_000
    }
}

public struct RealtimeConnectionState: Sendable {
    public private(set) var isOpening = false
    public private(set) var isConnected = false

    public init() {}

    public mutating func beginOpening() {
        isOpening = true
        isConnected = false
    }

    public mutating func confirmOpen() {
        isOpening = false
        isConnected = true
    }

    public mutating func disconnect() {
        isOpening = false
        isConnected = false
    }
}
```

private func confirmConnection(for task: URLSessionWebSocketTask) {

- [ ] **Step 5: Replace the three-second fallback loop**

In `MockAppStore`, keep a fallback attempt counter and use the policy:

```swift
private let realtimeFallbackPolicy = RealtimeFallbackPolicy()
private var fallbackSyncAttempt = 0

func startBackendAutoRefresh() {
    guard isBackendEnabled, let backendToken else { return }
    startRealtimeConnection(token: backendToken)
    guard backendRefreshTask == nil else { return }
    backendRefreshTask = Task { [weak self] in
        while let self, !Task.isCancelled {
            let delay = self.realtimeFallbackPolicy.delayNanoseconds(forAttempt: self.fallbackSyncAttempt)
            try? await Task.sleep(nanoseconds: delay)
            guard !Task.isCancelled else { break }
            guard !self.isRealtimeConnected else {
                self.fallbackSyncAttempt = 0
                continue
            }
            let succeeded = await self.syncFromBackend(reason: "fallback")
            self.fallbackSyncAttempt = succeeded ? 0 : min(self.fallbackSyncAttempt + 1, 3)
        }
    }
}
```

Make both `refreshFromBackend` and `syncFromBackend` return `Bool` with `@discardableResult`: return `true` after applying a bootstrap or sync payload, return the bootstrap result when no cursor exists, and return `false` for non-auth failures. Reset `fallbackSyncAttempt` after a confirmed socket connection.

- [ ] **Step 3: Implement the smoke script**

```js
import { pathToFileURL } from "node:url";
import WebSocket from "ws";

export function websocketHeaders(token) {
  if (!token) throw new Error("WS_TEST_TOKEN is required");
  return { Authorization: `Bearer ${token}` };
}

export async function verifyWebSocket(url, token, timeoutMs = 10000) {
  return new Promise((resolve, reject) => {
    const socket = new WebSocket(url, { headers: websocketHeaders(token) });
    const timer = setTimeout(() => {
      socket.terminate();
      reject(new Error("WebSocket handshake timed out"));
    }, timeoutMs);
    socket.once("open", () => {
      clearTimeout(timer);
      socket.close(1000, "verified");
      resolve({ ok: true, url });
    });
    socket.once("unexpected-response", (_request, response) => {
      clearTimeout(timer);
      reject(new Error(`WebSocket handshake returned HTTP ${response.statusCode}`));
    });
    socket.once("error", (error) => {
      clearTimeout(timer);
      reject(error);
    });
  });
}

if (import.meta.url === pathToFileURL(process.argv[1]).href) {
  const url = process.argv[2];
  if (!url) throw new Error("Usage: npm run verify:websocket -- <wss-url>");
  const result = await verifyWebSocket(url, process.env.WS_TEST_TOKEN || "");
  console.log(JSON.stringify(result));
}
```
